Pendiente:

Generar un grid el cual debe ser llenado con la siguiente consulta:

Select idUsuario, claveUsuario, Isnull(nombres, char(32)) Nombres, primerApellido, Isnull(segundoApellido, char(32)) segundoApellido, 
       b.descripcion estatus, c.descripcion tipoUsuario
From   dbo.segUsuariosTbl a
Join   dbo.catGeneralesTbl b
On     b.tabla = 'segUsuariosTbl'
And    b.columna = 'idEstatus'
And    b.valor   = a.idEstatus
Join   dbo.catGeneralesTbl c
On     c.tabla = b.tabla
And    c.columna = 'idTipoUsuario'
And    c.valor   = a.idTipoUsuario
Order by idUsuario



Esta son las reglas iniciales que se van a generar.

En frmConexion en la consulta a la tabla de segUsuariosTbl se le agrega el campo idTipoUsuario

                    string sqlUsuario = "Select  idUsuario, idEstatus As Bloqueado, idTipoUsuario " +
                                        "From    dbo.segUsuariosTbl " +
                                        "Where   ClaveUsuario = @ClaveUsuario ";

El fondo del grid LightSteelBlue

ClaveUsuario String(40) Alineado a la izquierda

Nombres String(100) Alineado a la izquierda

PrimerApellido String(100) Alineado a la izquierda

segundoApellido String(100) Alineado a la izquierda

Estatus String(20) Alineado al Centro

tipoUsuario String(100) Alineado a la izquierda

El boton de exportacion a excel (BtnExcel), exportara los datos plasmados en el grid.

Se usara el servicio ExcelExportService.cs  actualizando el código siguiente adaptandolo a nuestra aplicación

using ClosedXML.Excel;
using ClosedXML.Excel.Drawings;
using DocumentFormat.OpenXml.ExtendedProperties;
using RepGas.Infrastructure;
using System;
using System.Drawing;
using System.Drawing.Imaging;
using System.IO;
using System.Linq;
using System.Windows.Forms;


namespace SCMBD.Services
{
    public class ExcelExportService
    {
        private readonly IExcelExportRepository _excelExportRepository;
        public ExcelExportService(IExcelExportRepository excelExportRepository)
        {
            _excelExportRepository = excelExportRepository;
        }
        public void ExportarUsuarios(string archivo,DataGridView grid, string operacion, string usuario)
        {
            using (XLWorkbook workbook =
                new XLWorkbook())
            {
                var worksheet =
                    workbook.Worksheets.Add("Usuarios");

                AgregarLogo(worksheet, AppSettings.Usuario);

                ConstruirEncabezado(
                    worksheet,
                    operacion,
                    usuario);

                ConstruirDetalle(
                    worksheet,
                    grid);

                workbook.SaveAs(archivo);
            }
        }

        private void AgregarLogo(IXLWorksheet ws, string ciaCodigo)
        {
            string nombreLogo = $"logo.Trim().ToLower()}";

            Image logo = Properties.Resources.ResourceManager
                                .GetObject(nombreLogo) as Image;



            if (logo == null)
            {
                MessageBox.Show(
                    $"No existe el recurso: {nombreLogo}");
                return;
            }

            using (var ms = new MemoryStream())
            {
                logo.Save(ms, ImageFormat.Png);

                ms.Position = 0;

                var picture =
                    ws.AddPicture(
                        ms,
                        XLPictureFormat.Png,
                        "Logo");

                picture.MoveTo(ws.Cell("A1"));

                picture.WithSize(226, 85);
            }
        }

        private void ConstruirEncabezado(
     IXLWorksheet ws,
     int mes,
     int anio,
     string operacion,
     string usuario)
        {
            ws.Cell("E1").Value = "Página:";
            ws.Cell("E2").Value = "Fecha:";
            ws.Cell("E3").Value = "Reporte:";
            ws.Cell("E4").Value = "Usuario:";

            ws.Range("E1:E4")
              .Style.Font.SetBold(true);

            ws.Cell("F1").Value = "1";

            ws.Cell("F2").Value = DateTime.Now;

            ws.Cell("F2").Style.NumberFormat.Format =
                "dd/MM/yyyy HH:mm";

            ws.Cell("F3").Value = operacion;

            ws.Cell("F4").Value = usuario;

            ws.Range("A6:F6").Merge();

            ws.Cell("A6").Value =
                $"Reporte Usuarios relacionados en SCMBD";

            ws.Cell("A6").Style.Font.Bold = true;
            ws.Cell("A6").Style.Font.FontSize = 12;
            ws.Cell("A6").Style.Font.Underline =
                XLFontUnderlineValues.Single;

            ws.Cell("A6").Style.Alignment.Horizontal =
                XLAlignmentHorizontalValues.Center;
        }

        public void ExportToExcel(string filePath, string sheetName, DataGridView dataGridView)
        {
            using (var workbook = new XLWorkbook())
            {
                var worksheet = workbook.Worksheets.Add(sheetName);
                // Add column headers
                for (int i = 0; i < dataGridView.Columns.Count; i++)
                {
                    worksheet.Cell(1, i + 1).Value = dataGridView.Columns[i].HeaderText;
                }
                // Add rows
                for (int i = 0; i < dataGridView.Rows.Count; i++)
                {
                    for (int j = 0; j < dataGridView.Columns.Count; j++)
                    {
                        worksheet.Cell(i + 2, j + 1).Value = dataGridView.Rows[i].Cells[j].Value?.ToString();
                    }
                }
                // Save the workbook to the specified file path
                workbook.SaveAs(filePath);
            }
        }
    }
}
